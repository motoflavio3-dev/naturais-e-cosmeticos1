function processarImagem(file) {
if (file.size > 2 * 1024 * 1024) { mostrarToast('❌ Imagem muito grande! Máximo 2MB'); return; }
if (!file.type.startsWith('image/')) { mostrarToast('❌ Selecione apenas imagens!'); return; }
const reader = new FileReader();
reader.onload = function(e) {
const img = new Image();
img.onload = function() {
const canvas = document.createElement('canvas');
let width = img.width, height = img.height;
const MAX_SIZE = 800;
if (width > height && width > MAX_SIZE) { height = (height * MAX_SIZE) / width; width = MAX_SIZE; }
else if (height > MAX_SIZE) { width = (width * MAX_SIZE) / height; height = MAX_SIZE; }
canvas.width = width; canvas.height = height;
const ctx = canvas.getContext('2d');
ctx.drawImage(img, 0, 0, width, height);
const base64 = canvas.toDataURL('image/jpeg', 0.8);
imagemAtual = base64;
document.getElementById('pImagem').value = base64;
document.getElementById('previewImg').src = base64;
document.getElementById('uploadPreview').classList.add('active');
document.getElementById('uploadArea').style.display = 'none';
};
img.src = e.target.result;
};
reader.readAsDataURL(file);
}

function getTodasCategorias() {
const catsProdutos = [...new Set(produtos.map(p => p.categoria))];
return [...new Set([...categoriasCustom, ...catsProdutos])].sort((a, b) => a.localeCompare(b));
}

function contarProdutosCategoria(categoria) {
return produtos.filter(p => p.categoria === categoria).length;
}

function adicionarCategoria() {
const input = document.getElementById('novaCategoriaInput');
const nome = input.value.trim();
if (!nome) { mostrarToast('❌ Digite o nome da categoria!'); return; }
const todas = getTodasCategorias();
if (todas.some(c => c.toLowerCase() === nome.toLowerCase())) { mostrarToast('❌ Esta categoria já existe!'); return; }
categoriasCustom.push(nome);
salvarCategorias();
input.value = '';
renderCategoriasLista();
renderCategorias();
mostrarToast('✅ Categoria adicionada!');
}

function editarCategoria(nomeAtual) {
const container = document.getElementById('cat-item-' + CSS.escape(nomeAtual));
if (!container) return;
container.innerHTML = `<div class="cat-edit-form"><input type="text" id="edit-cat-input" value="${nomeAtual}" maxlength="30"><button class="cat-save-btn" onclick="salvarEdicaoCategoria('${nomeAtual.replace(/'/g, "\\'")}')">✓</button><button class="cat-cancel-btn" onclick="renderCategoriasLista()">✗</button></div>`;
const input = document.getElementById('edit-cat-input');
input.focus(); input.select();
input.addEventListener('keypress', (e) => { if (e.key === 'Enter') { e.preventDefault(); salvarEdicaoCategoria(nomeAtual); } });

function salvarEdicaoCategoria(nomeAtual) {
const input = document.getElementById('edit-cat-input');
const novoNome = input.value.trim();
if (!novoNome) { mostrarToast('❌ Nome não pode estar vazio!'); return; }
if (novoNome === nomeAtual) { renderCategoriasLista(); return; }
const todas = getTodasCategorias();
if (todas.some(c => c.toLowerCase() === novoNome.toLowerCase() && c !== nomeAtual)) { mostrarToast('❌ Já existe uma categoria com este nome!'); return; }
const idx = categoriasCustom.indexOf(nomeAtual);
if (idx !== -1) categoriasCustom[idx] = novoNome; else categoriasCustom.push(novoNome);
produtos.forEach(p => { if (p.categoria === nomeAtual) p.categoria = novoNome; });
if (categoriaAtiva === nomeAtual) categoriaAtiva = novoNome;
salvarCategorias(); salvarProdutos();
renderCategoriasLista(); renderCategorias(); renderProdutos();
mostrarToast('✅ Categoria renomeada!');
}

function excluirCategoria(nome) {
const qtd = contarProdutosCategoria(nome);
let msg = `Excluir a categoria "${nome}"?`;
if (qtd > 0) msg += `\n⚠️ Existem ${qtd} produto(s). OK = Excluir tudo, Cancelar = Manter`;
if (!confirm(msg)) return;
categoriasCustom = categoriasCustom.filter(c => c !== nome);
if (qtd > 0) { produtos = produtos.filter(p => p.categoria !== nome); salvarProdutos(); }
if (categoriaAtiva === nome) categoriaAtiva = 'Todos';
salvarCategorias(); renderCategoriasLista(); renderCategorias(); renderProdutos(); renderAdminLista();
mostrarToast('🗑️ Categoria excluída!');
}

e.preventDefault();
const id = document.getElementById('produtoId').value;
let imagemFinal = document.getElementById('pImagem').value || '🌿';
const novaCategoria = document.getElementById('pCategoria').value.trim();
const novo = { nome: document.getElementById('pNome').value.trim(), categoria: novaCategoria, preco: parseFloat(document.getElementById('pPreco').value), desc: document.getElementById('pDesc').value.trim(), imagem: imagemFinal };
const todas = getTodasCategorias();
if (!todas.some(c => c.toLowerCase() === novaCategoria.toLowerCase())) { categoriasCustom.push(novaCategoria); salvarCategorias(); }
if (id) { const idx = produtos.findIndex(p => p.id === parseInt(id)); produtos[idx] = { ...produtos[idx], ...novo }; mostrarToast('✅ Produto atualizado!'); }
else { const novoId = produtos.length > 0 ? Math.max(...produtos.map(p => p.id)) + 1 : 1; produtos.push({ id: novoId, ...novo }); mostrarToast('✅ Produto adicionado!'); }
salvarProdutos(); fecharForm(); renderAdminLista(); renderCategorias(); renderCategoriasLista(); renderProdutos();
}

function excluirProduto(id) {
if (!confirm('Tem certeza que deseja excluir este produto?')) return;
produtos = produtos.filter(p => p.id !== id);
salvarProdutos(); renderAdminLista(); renderCategorias(); renderCategoriasLista(); renderProdutos();
mostrarToast('🗑️ Produto excluído!');
}

function mostrarToast(msg) {
const toast = document.getElementById('toast');
toast.textContent = msg; toast.classList.add('show');
setTimeout(() => toast.classList.remove('show'), 2500);
}
